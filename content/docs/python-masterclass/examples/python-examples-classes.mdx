Classes များသည် objects များကို မည်သို့ ဖန်တီး၍ data နှင့် behavior များကို စုစည်းကြောင်း သရုပ်ပြပါသည်.

## Simple Class

```
class Person:
    def __init__(self, name, age):
        self.name = name
        self.age = age

    def greet(self):
        return f"Hi, I'm {self.name}"

john = Person("John", 30)
print(john.greet())        # Hi, I'm John
print(john.age)            # 30
```

## Property with Getter/Setter

```
class BankAccount:
    def __init__(self, owner, balance=0):
        self.owner = owner
        self.__balance = balance   # private

    def deposit(self, amount):
        if amount > 0:
            self.__balance += amount

    def withdraw(self, amount):
        if 0 < amount <= self.__balance:
            self.__balance -= amount
            return amount
        return 0

    @property
    def balance(self):
        return self.__balance

account = BankAccount("John", 100)
account.deposit(50)
account.withdraw(30)
print(account.balance)   # 120
```

## Inheritance

```
class Animal:
    def __init__(self, name):
        self.name = name

    def speak(self):
        return "..."

class Dog(Animal):
    def speak(self):
        return "Woof!"

class Cat(Animal):
    def speak(self):
        return "Meow!"

for animal in [Dog("Rex"), Cat("Milo")]:
    print(f"{animal.name}: {animal.speak()}")
```

## Class Variables vs Instance Variables

```
class Employee:
    company = "TechCorp"   # class variable

    def __init__(self, name, salary):
        self.name = name    # instance variable
        self.salary = salary

e1 = Employee("John", 4000)
e2 = Employee("Amy", 3500)
print(e1.company, e2.company)   # TechCorp TechCorp
print(Employee.company)         # TechCorp
```

## Magic Methods

```
class Point:
    def __init__(self, x, y):
        self.x = x
        self.y = y

    def __repr__(self):
        return f"Point({self.x}, {self.y})"

    def __add__(self, other):
        return Point(self.x + other.x, self.y + other.y)

    def __eq__(self, other):
        return self.x == other.x and self.y == other.y

p1 = Point(1, 2)
p2 = Point(3, 4)
print(p1 + p2)      # Point(4, 6)
print(p1 == Point(1, 2))  # True
```

## Composition

```
class Engine:
    def start(self):
        return "Engine started"

class Car:
    def __init__(self, model):
        self.model = model
        self.engine = Engine()   # composition

    def start(self):
        return f"{self.model}: {self.engine.start()}"

car = Car("Toyota")
print(car.start())   # Toyota: Engine started
```

## Polymorphism

```
class Shape:
    def area(self):
        return 0

class Rectangle(Shape):
    def __init__(self, w, h):
        self.w = w
        self.h = h

    def area(self):
        return self.w * self.h

class Circle(Shape):
    def __init__(self, radius):
        self.radius = radius

    def area(self):
        import math
        return math.pi * self.radius**2

shapes = [Rectangle(4, 5), Circle(3)]
for shape in shapes:
    print(shape.area())
# 20
# 28.27...
```
