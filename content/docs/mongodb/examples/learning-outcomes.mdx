## ‌စစ်ဆေး — ‌ဥပမာ

```js
// Competency check — ‌ပြည့် ‌ဖို့
use checkdb;

db.orders.insertMany([
  { user: "u1", item: "A", qty: 2, total: 20 },
  { user: "u2", item: "B", qty: 1, total: 5 },
  { user: "u1", item: "C", qty: 1, total: 7 }
]);

// Find ‌အား‌လုံး — total > 15
db.orders.find({ total: { $gt: 15 } });

// ‌အသုံးပြု‌သူ အား‌လုံး — ‌ပေါင်း
db.orders.aggregate([
  { $group: { _id: "$user", spent: { $sum: "$total" } } }
]);
```
